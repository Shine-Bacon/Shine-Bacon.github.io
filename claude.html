📝 常用 SQL 指令
-- 查詢分數排行
SELECT m.num, m.name, s.score
FROM members m JOIN scores s ON m.num = s.member_num
ORDER BY s.score DESC;

-- 加分並記錄
UPDATE scores SET score = score + 10 WHERE member_num = 1;
INSERT INTO score_log (member_num, delta, reason) VALUES (1, 10, '上課積極');

-- 新增作業
INSERT INTO homeworks (subject, title, due_date)
VALUES ('數學', '第六單元', '2026-03-10');

-- 查詢未完成作業
SELECT subject, title, due_date FROM homeworks
WHERE is_done = 0 ORDER BY due_date;

-- 查抽籤紀錄
SELECT m.name, d.drawn_at FROM draw_history d
JOIN members m ON d.member_num = m.num
ORDER BY d.drawn_at DESC;
